<script>
const MEMSIZE = 8000;
const mem = new Int32Array(MEMSIZE);
let sp = 0;

// Registres d'execució de les 9+ primitives
let keyBuffer = [];
let compiling = false;
let currentWord = null;
let keyResolver = null;
let outBuffer = "";

// Diccionari compilat
// Tipus d'instrucció compilada:
// 0: Literal (val = número)
// 1..14: Primitives natives
// 15: Crida a paraula definida (val = índex a `words`)
const OP_LIT     = 0;
const OP_ADD     = 1;
const OP_NAND    = 2;
const OP_DROP    = 3;
const OP_ZEQ     = 4;
const OP_FETCH   = 5;
const OP_STORE   = 6;
const OP_SPFETCH = 7;
const OP_EMIT    = 8;
const OP_DOT     = 9;
const OP_KEY     = 10;
const OP_KEYQ    = 11;
const OP_MS      = 12;
const OP_BRANCH  = 13;
const OP_QBRANCH = 14;
const OP_CALL    = 15;

const PRIM_MAP = {
  "+": OP_ADD, "nand": OP_NAND, "drop": OP_DROP, "0=": OP_ZEQ,
  "@": OP_FETCH, "!": OP_STORE, "sp@": OP_SPFETCH, "emit": OP_EMIT,
  ".": OP_DOT, "key": OP_KEY, "key?": OP_KEYQ, "ms": OP_MS,
  "branch": OP_BRANCH, "?branch": OP_QBRANCH
};

let codeOp = [];   // Tipus d'operació
let codeVal = [];  // Valor associat (literal o índex de paraula)
let words = [];
let wordMap = new Map();

const con = document.getElementById("console");
const inp = document.getElementById("input");
const btn = document.getElementById("btn");
const stk = document.getElementById("stack");
const wrdList = document.getElementById("words");

function flushOut() {
  if (outBuffer.length > 0) {
    con.textContent += outBuffer;
    outBuffer = "";
    con.scrollTop = con.scrollHeight;
  }
}

function out(t) {
  outBuffer += t;
  if (t.indexOf("\n") !== -1) flushOut();
}

function prompt() {
  out("ok> ");
  flushOut();
}

function updateStack() {
  if (sp === 0) {
    stk.textContent = "Pila (0): [buida]";
    return;
  }
  let items = new Array(sp);
  for (let i = 0; i < sp; i++) items[i] = mem[i];
  stk.textContent = "Pila (" + sp + "): " + items.join(" ") + " <-top";
}

function updateWordsList() {
  if (words.length === 0) {
    wrdList.textContent = "Paraules Creades: [cap]";
    return;
  }
  wrdList.textContent = "Paraules Creades: " + words.map(w => w.name).join(" ");
}

function push(x) {
  if (sp >= MEMSIZE) throw "stack overflow";
  mem[sp++] = x | 0;
}

function pop() {
  if (sp <= 0) throw "stack underflow";
  return mem[--sp];
}

function compileToken(tok) {
  if (/^-?\d+$/.test(tok)) {
    return { op: OP_LIT, val: parseInt(tok, 10) | 0 };
  }
  if (PRIM_MAP[tok] !== undefined) {
    return { op: PRIM_MAP[tok], val: 0 };
  }
  let wIdx = wordMap.get(tok);
  if (wIdx !== undefined) {
    return { op: OP_CALL, val: wIdx };
  }
  throw "unknown word: " + tok;
}

// Execució ultraràpida sense async/await excepte quan es crida key o ms
async function runCompiled(start, end) {
  let ip = start;
  let a, b, addr, flag;

  while (ip < end) {
    let op = codeOp[ip];
    let val = codeVal[ip];
    ip++;

    switch (op) {
      case OP_LIT:
        if (sp >= MEMSIZE) throw "stack overflow";
        mem[sp++] = val;
        break;
      case OP_ADD:
        b = pop(); a = pop(); push(a + b);
        break;
      case OP_NAND:
        b = pop(); a = pop(); push(~(a & b));
        break;
      case OP_DROP:
        pop();
        break;
      case OP_ZEQ:
        a = pop(); push(a === 0 ? -1 : 0);
        break;
      case OP_FETCH:
        addr = pop();
        if (addr < 0 || addr >= MEMSIZE) throw "bad address";
        push(mem[addr]);
        break;
      case OP_STORE:
        addr = pop(); a = pop();
        if (addr < 0 || addr >= MEMSIZE) throw "bad address";
        mem[addr] = a;
        break;
      case OP_SPFETCH:
        push(sp - 1);
        break;
      case OP_EMIT:
        out(String.fromCharCode(pop() & 255));
        break;
      case OP_DOT:
        out(pop() + " ");
        break;
      case OP_KEY:
        flushOut();
        if (keyBuffer.length === 0 && inp.value.length === 0) {
          await new Promise(r => keyResolver = r);
        }
        if (keyBuffer.length > 0) {
          push(keyBuffer.shift());
        } else {
          let c = inp.value.charCodeAt(0);
          inp.value = inp.value.slice(1);
          push(c);
        }
        break;
      case OP_KEYQ:
        push((keyBuffer.length > 0 || inp.value.length > 0) ? -1 : 0);
        break;
      case OP_MS:
        flushOut();
        await new Promise(r => setTimeout(r, pop()));
        break;
      case OP_BRANCH:
        addr = pop();
        ip = start + addr;
        break;
      case OP_QBRANCH:
        addr = pop(); flag = pop();
        if (flag === 0) ip = start + addr;
        break;
      case OP_CALL:
        let w = words[val];
        await runCompiled(w.start, w.end);
        break;
    }
  }
}

async function interpret(line) {
  let clean = line.replace(/\u00A0/g, " ");
  clean = clean.replace(/\(\s[^\)]*\)/g, " ");
  let idx = clean.indexOf("\\");
  if (idx >= 0) clean = clean.substring(0, idx);
  clean = clean.trim();
  if (clean === "") return;

  let tokens = clean.split(/\s+/);
  for (let i = 0; i < tokens.length; i++) {
    let tok = tokens[i];
    if (!compiling) {
      if (tok === ":") {
        i++;
        if (i >= tokens.length) throw "missing name";
        currentWord = { name: tokens[i], start: codeOp.length, end: 0 };
        compiling = true;
        continue;
      }
      let inst = compileToken(tok);
      if (inst.op === OP_BRANCH || inst.op === OP_QBRANCH) {
        throw tok + " outside definition";
      }
      // Executem una instrucció directa
      let tmpStart = codeOp.length;
      codeOp.push(inst.op);
      codeVal.push(inst.val);
      try {
        await runCompiled(tmpStart, tmpStart + 1);
      } finally {
        codeOp.length = tmpStart;
        codeVal.length = tmpStart;
      }
      continue;
    }

    if (tok === ";") {
      currentWord.end = codeOp.length;
      let newIdx = words.length;
      words.push(currentWord);
      wordMap.set(currentWord.name, newIdx);
      compiling = false;
      currentWord = null;
      continue;
    }

    let inst = compileToken(tok);
    codeOp.push(inst.op);
    codeVal.push(inst.val);
  }
  flushOut();
}

async function executar() {
  let line = inp.value;
  if (keyResolver) {
    for (let i = 0; i < line.length; i++) keyBuffer.push(line.charCodeAt(i));
    keyBuffer.push(13);
    inp.value = "";
    let res = keyResolver;
    keyResolver = null;
    res();
    return;
  }
  inp.value = "";
  out(line + "\n");
  try {
    await interpret(line);
  } catch (e) {
    out("ERROR: " + e + "\n");
  }
  flushOut();
  inp.value = "";
  updateStack();
  updateWordsList();
  if (!keyResolver) prompt();
  inp.focus();
}

btn.onclick = executar;
inp.oninput = function() {
  if (keyResolver && inp.value.length > 0) {
    let res = keyResolver;
    keyResolver = null;
    res();
  }
};
inp.onkeydown = function(e) {
  if (e.key === "Enter" || e.keyCode === 13) {
    e.preventDefault();
    executar();
  }
};

out("MiniForth\n\n");
prompt();
</script>
